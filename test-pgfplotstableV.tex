\usepackage{pgfplotstable}
\usepackage{booktabs}
\usepackage{xparse}


% Create a ReplaceSuffix command
\makeatletter
\def\ReplaceSuffix#1#2{%
  \filename@parse{#1}%
  \edef\ReplaceSuffixResult{\filename@base.#2}%
}
\makeatother

% Set pgfplotstable
\pgfplotsset{compat=1.18}
\pgfplotstableset{row sep=newline,col sep=comma}

\newcommand{\gobblearg}[1]{}

\def\saveTable{1}

\DeclareDocumentCommand{\includepgfplotstable}{>{\TrimSpaces}m m m m m m}{
\centering
\begin{minipage}{\textwidth}
\pgfplotstabletypeset[
  header=false,
  skip first n=1,
  #1,
  #2,
  begin table={\begin{tabular}{#3}
    \toprule
    #4
    \midrule\gobblearg},  
  end table={
    \bottomrule
    \end{tabular}
  },
  every head row/.style={output empty row, after row={}},
  every last row/.style={after row={}}
]{#5}
#6
\end{minipage}
}

\ifdefined\saveTable

\DeclareDocumentCommand{\includetable}{>{\TrimSpaces}m m m m m}{
\ReplaceSuffix{#4}{tex}
\edef\outfiletemp{\ReplaceSuffixResult}%
\includepgfplotstable{typeset=true,output=\outfiletemp}{#1}{#2}{#3}{#4}{#5}
}

\else

\DeclareDocumentCommand{\includetable}{>{\TrimSpaces}m m m m m}{
\includepgfplotstable{}{#1}{#2}{#3}{#4}{#5}
}

\fi
